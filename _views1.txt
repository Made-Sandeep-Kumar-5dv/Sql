1. Create a view named employee_details that displays all employees.
create view employee_details as select * from employees;

2. Create a view named it_employees that displays only employees working in the IT department.
create view it_employees as select emp_name from employees where department = ‘IT’;

3. Create a view named high_salary_employees that displays employees whose salary is greater than 50000.
create view high_salary_employees as select emp_name from employees where salary > 50000;

4. Create a view that displays only emp_name, department, and salary.
create view emp as select emp_name,department,salary from employees;

5. Create a view that displays employees from Hyderabad.
create view city_employees as select * from employees where city = ‘Hyderabad’;

6. Display the records from the it_employees view.
select * from it_employees;

7. Display only the employee name and salary from the employee_details view.
create or replace view employee_details as select emp_name,salary from employees;

8. Create a view that displays employees whose salary is between 40000 and 55000.
create view sal as select * from employees where salary between 40000 and 55000;

9. Create a view named hr_employees containing employees from the HR department. Then retrieve all records from it.
create view hr_employees as select emp_id,emp_name from employees where department = ‘HR’;

10. Create a view that displays the department and average salary for each department.
create view avg_salary as select department,avg(salary) from employees group by department;

11. Create a view that displays the department and maximum salary in each department.
create view max_salary as select department,max(salary) from employees group by department;

12. Create a view that displays employees in descending order of salary. Then retrieve the data from the view.
Create view sal select *from employees order by salary desc;

13. Create a view that contains employees whose names start with the letter A.
create view ab as select * from employees where emp_name like ‘A%’;

14. Create a view that contains employees from either Chennai or Hyderabad.
create view city as select * from employees where city in (‘Chennai’,’Hyderabad’);


15. Create a view named employee_salary with emp_id, emp_name, and salary. Then increase the salary of employee 101 through the view and check whether the base table is affected.
create view employee_salary as select emp_id,emp_name,salary from employees;
update employee_salary set salary = salary + 10000 where emp_id = 101;
select * from employee_salary;
